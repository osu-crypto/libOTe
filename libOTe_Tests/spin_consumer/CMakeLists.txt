cmake_minimum_required(VERSION 3.20)
project(libote_spin_consumer LANGUAGES CXX)
find_package(libOTe CONFIG REQUIRED COMPONENTS spin)
if(NOT libOTe_spin_FOUND OR NOT TARGET spin::spin)
    message(FATAL_ERROR "libOTe package does not expose its SPIN dependency")
endif()
add_executable(spin_consumer main.cpp)
target_link_libraries(spin_consumer PRIVATE oc::libOTe)
enable_testing()
add_test(NAME SpinConsumer COMMAND spin_consumer)
set_tests_properties(SpinConsumer PROPERTIES SKIP_RETURN_CODE 77)
